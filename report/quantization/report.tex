\section{Research Design}

\subsection{Research Questions}

\begin{itemize}
\item \textbf{RQ1:} Does quantization affect RAG answer accuracy compared to parametric knowledge?
\item \textbf{RQ2:} Does quantization increase hallucination rates and reduce context faithfulness?
\item \textbf{RQ3:} How does quantization impact performance across varying context lengths?
\item \textbf{RQ4:} Does quantization affect output consistency across multiple runs?
\item \textbf{RQ5:} How does quantization affect robustness to distractor passages?
\end{itemize}

\textbf{Quantization Details:}
\begin{itemize}
\item \textbf{AWQ:} 4-bit weights, group size 128, zero-point quantization, per-group scaling
\item \textbf{GPTQ:} 4-bit weights, group size 128, asymmetric quantization
\item \textbf{NF4:} 4-bit NormalFloat, double quantization, on-the-fly via BitsAndBytes
\item \textbf{Compression:} All quantized methods achieve 4.32$\times$ compression vs FP16
\end{itemize}

\begin{itemize}

\item Attention heads: 32 (128-dim each)
\item FFN intermediate: 14,336
\item Vocabulary: 32,000 (SentencePiece)
\item Max sequence length: 4096 tokens
\end{itemize}

\section{Prompt Format Selection Study}

\subsection{Design}

\textbf{Purpose:} Identify the optimal prompt format for base and instruction-tuned model variants across all quantization methods.

\textbf{Evaluated Formats:}

\textit{Instruction-tuned models (5 formats):}
\begin{itemize}
\item \textbf{Minimal:} \texttt{[INST] \{context\}\textbackslash n\textbackslash n\{question\} [/INST]}
\item \textbf{Verbose:} Explicit instructions with "Answer:" label
\item \textbf{Chat:} \texttt{[INST] Context: \{context\}\textbackslash n\textbackslash nBased on this, \{question\} [/INST]}
\item \textbf{System:} Includes \texttt{<<SYS>>} block with role description
\item \textbf{QA:} \texttt{[INST] Context: \{context\}\textbackslash n\textbackslash nQ: \{question\}\textbackslash nA: [/INST]}
\end{itemize}

\textit{Base models (5 formats):}
\begin{itemize}
\item \textbf{Minimal:} \texttt{\{context\}\textbackslash n\textbackslash nQ: \{question\}\textbackslash nA:}
\item \textbf{Few-shot (1 example):} Single demonstration before target question
\item \textbf{Few-shot (2 examples):} Two demonstrations before target question
\item \textbf{Labeled:} \texttt{Context: \{context\}\textbackslash n\textbackslash nQuestion: \{question\}\textbackslash nAnswer:}
\item \textbf{Completion:} \texttt{\{context\}\textbackslash n\textbackslash nThe answer to "\{question\}" is:}
\end{itemize}

\textbf{Test Set:}
\begin{itemize}
\item 30 questions stratified across 6 categories (5 per category)
\item Categories: simple factual, multi-hop, numerical, context-dependent, unanswerable, long-form
\item Sources: SQuAD v2 (validation), HotpotQA (validation), DROP (validation), NarrativeQA (validation), ELI5 (validation)
\item \textbf{Total:} 8 configs $\times$ 5 formats $\times$ 30 questions = 1,200 generations
\end{itemize}

\textbf{Evaluation Metrics:}
\begin{itemize}
\item \textbf{Pass/Fail Binary:} Answer correctness + output quality checks
\item Quality checks: token loops, excessive special characters, reasonable length, lexical diversity
\item Answer matching: fuzzy string matching (rapidfuzz partial\_ratio $\geq$ 0.85) + exact substring matching
\end{itemize}

\section{Set A: Answer Quality \& Faithfulness}

\subsection{Design}

\textbf{Purpose:} Measure accuracy and hallucination rates (RQ1, RQ2)

\textbf{Sample Composition:}
\begin{itemize}
\item Factual questions: 30 (SQuAD v2)
\item Multihop questions: 25 (HotpotQA)
\item Numerical questions: 20 (DROP)
\item Unanswerable questions: 25 (SQuAD v2)
\item \textbf{Total:} 100 samples
\end{itemize}

\textbf{Conditions per sample:}
\begin{itemize}
\item RAG: Question + context (1024 tokens max)
\item No-RAG: Question only (parametric knowledge)
\item \textbf{Total generations:} 100 samples $\times$ 2 conditions = 200 per model
\end{itemize}

\textbf{Prompt Templates:}

\textit{Base Models - RAG:}
\begin{lstlisting}
{context}\n\nQ: {question}\nA:
\end{lstlisting}

\textit{Base Models - No-RAG:}
\begin{lstlisting}
Q: {question}\nA:
\end{lstlisting}

\textit{Instruction Models - RAG:}
\begin{lstlisting}
[INST] {context}\n\n{question} [/INST]
\end{lstlisting}

\textit{Instruction Models - No-RAG:}
\begin{lstlisting}
[INST] {question} [/INST]
\end{lstlisting}

\subsection{Metrics}

\textbf{Accuracy Metrics:}
\begin{itemize}
\item \textbf{Exact Match (EM):} Binary string equality after normalization
\item \textbf{Token F1:} Token-level precision/recall, max over ground truth variants
\item \textbf{Substring Match:} Whether ground truth appears in prediction
\item \textbf{Token Precision:} $\frac{|\text{pred} \cap \text{ref}|}{|\text{pred}|}$
\item \textbf{Token Recall:} $\frac{|\text{pred} \cap \text{ref}|}{|\text{ref}|}$
\item \textbf{Jaccard Similarity:} $\frac{|\text{pred} \cap \text{ref}|}{|\text{pred} \cup \text{ref}|}$
\end{itemize}

\textbf{Normalization Protocol (SQuAD-style):}
\begin{lstlisting}[language=Python]
def normalize_answer(text):
    text = unicodedata.normalize('NFD', text).lower()
    text = re.sub(r'\b(a|an|the)\b', ' ', text)
    text = re.sub(r'[^a-z0-9\s]', '', text)
    return ' '.join(text.split()).strip()
\end{lstlisting}

\textbf{Extraction Strategies:}
\begin{itemize}
\item \textbf{Raw:} Minimal processing (prompt removal only)
\item \textbf{Conservative:} First line + length-based truncation (3$\times$ ground truth)
\item \textbf{Aggressive:} First sentence + first line
\end{itemize}

\textbf{Faithfulness Metrics:}
\begin{itemize}
\item \textbf{Token Overlap:} $\frac{|\text{pred tokens} \cap \text{context tokens}|}{|\text{pred tokens}|}$
\item \textbf{Hallucination Score:} $1 - \text{Token Overlap}$
\item \textbf{Bigram Attribution:} Fraction of prediction bigrams in context
\item \textbf{Trigram Attribution:} Fraction of prediction trigrams in context
\item \textbf{Context Sufficiency:} Same as token overlap (simple heuristic)
\end{itemize}

\textbf{Statistical Analysis:}
\begin{itemize}
\item Paired t-tests (RAG vs No-RAG)
\item Bootstrap 95\% CI (1000 iterations)
\item Cohen's d effect sizes
\item Bonferroni correction: $\alpha = 0.05/8 = 0.00625$
\end{itemize}

\section{Set B: Context Length Robustness}

\subsection{Design}

\textbf{Purpose:} Test performance degradation across context lengths (RQ3)

\textbf{Experimental Design:}
\begin{itemize}
\item \textbf{Context Lengths:} 512, 1024, 2048, 4096 tokens
\item \textbf{Answer Positions:} start, middle, end
\item \textbf{Samples per config:} 20
\item \textbf{Total:} 4 lengths $\times$ 3 positions $\times$ 20 samples = 240 per model
\end{itemize}

\textbf{Context Padding Protocol:}
\begin{lstlisting}[language=Python]
# Base samples from SQuAD v2 (100-400 token contexts)
# Pad to target length using WikiText-103 distractor passages
# Place original context at specified position (start/middle/end)
# Verify answer position post-padding (within 25% tolerance)

def pad_context(original_ctx, target_length, position):
    distractor = sample_from_wikitext103()
    if position == 'start':
        return original_ctx + distractor
    elif position == 'middle':
        mid = len(distractor) // 2
        return distractor[:mid] + original_ctx + distractor[mid:]
    else:  # end
        return distractor + original_ctx
\end{lstlisting}

\textbf{Prompt:} RAG prompts only (same as Set A)

\item \textbf{Entropy:} Shannon entropy of normalized outputs (lower = more consistent)
\end{itemize}

\textbf{Accuracy Stability:}
\begin{itemize}
\item Mean and standard deviation of Exact Match and F1 across 5 runs per sample
\item Lower standard deviation indicates more stable accuracy
\end{itemize}

\textbf{Statistical Analysis:}
\begin{itemize}

\section{Set D: Distractor Sensitivity}

\subsection{Design}

\textbf{Purpose:} Test robustness to irrelevant context passages (RQ5)

\textbf{Experimental Design:}
\begin{itemize}
\item 75 samples from SQuAD v2 (context 200-400 tokens)
\item 2 conditions per sample: clean vs distracted
\item \textbf{Total:} 75 samples $\times$ 2 conditions = 150 generations per model
\end{itemize}

\textbf{Distractor Injection:}
\begin{lstlisting}[language=Python]
# Split original context at 50% point
# Insert 3 distractor passages (200 chars each, WikiText-103)
# Structure: Dist1 + Ctx_Part1 + Dist2 + Ctx_Part2 + Dist3
# Final context ~ 3x original length

def inject_distractors(context):
    mid = len(context) // 2
    part1, part2 = context[:mid], context[mid:]
    d1, d2, d3 = sample_distractors(n=3, length=200)
    return d1 + part1 + d2 + part2 + d3
\end{lstlisting}

\textbf{Prompt:} RAG prompts only (same as Set A)

\subsection{Metrics}

\textbf{Accuracy Degradation:}
\begin{itemize}
\item \textbf{Accuracy Drop:} Clean F1 - Distracted F1
\item Paired t-test for statistical significance
\item Cohen's d effect size
\end{itemize}

\textbf{Distractor Detection:}
\begin{itemize}
\item \textbf{Citation Rate:} \% samples citing distractor content (token overlap $>$ 20\%)
\item \textbf{Distractor Token Overlap:} Fraction of prediction tokens in distractor-only text
\item \textbf{Clean Token Overlap:} Fraction of prediction tokens in original context
\end{itemize}

\textbf{Answer Consistency:}
\begin{itemize}
\item Jaccard similarity between clean and distracted predictions
\end{itemize}

\section{Efficiency Trade-offs Analysis}

\subsection{Design}

\textbf{Purpose:} Quantify latency, throughput, memory usage, and model size to identify optimal quality-efficiency trade-offs for RAG deployment (RQ5).

\textbf{Measurement Sources:}
\begin{itemize}
\item \textbf{Timing Data:} Extracted from generation timestamps across all evaluation sets (A, B, C, D)
\item \textbf{Benchmark Timing:} Standalone inference tests with fixed prompts (10 runs, 3 warmup)
\item \textbf{Memory:} GPU memory profiling (\texttt{torch.cuda.memory\_allocated}, \texttt{peak\_memory\_allocated})
\item \textbf{Model Size:} Calculated from parameter count and bits per parameter
\end{itemize}

\textbf{Generation Timing Extraction:}
\begin{lstlisting}[language=Python]
# Per-sample timing recorded during Sets A-D evaluation
for sample in data['samples']:
    times.append(sample['rag_generation_time_ms'])
    times.append(sample['no_rag_generation_time_ms'])

# Aggregate statistics across 590 generations per model
timing_stats = {
    'mean_ms': np.mean(times),
    'median_ms': np.median(times),
    'std_ms': np.std(times),
    'p95_ms': np.percentile(times, 95),
    'p99_ms': np.percentile(times, 99)
}

# Estimate throughput (assumes 64 avg tokens)
tokens_per_sec = (64 / mean_ms) * 1000
\end{lstlisting}

\section{Perplexity and Context Impact Evaluation}

\subsection{Design}

\textbf{Purpose:} Measure language modeling quality and context utilization capability

\textbf{Datasets:}
\begin{itemize}
\item WikiText-103 test set: 100 samples
\item C4 validation set (streamed): 100 samples
\item Sample filtering: minimum 100 characters, minimum 20 words, $\geq$ 50\% alphabetic ratio
\item Maximum context: 2048 tokens with 512-token stride
\end{itemize}

\textbf{Evaluation Components:}
\begin{enumerate}
\item \textbf{Baseline Perplexity:} Standard perplexity measurement on complete documents
\item \textbf{Context Impact:} Continuation prediction with/without preceding context
\end{enumerate}

\textbf{Perplexity Computation:}
\begin{lstlisting}[language=Python]
# Negative log-likelihood: NLL = -sum log P(x_i | x_<i)
# Token-weighted averaging across sequences
# Perplexity: PPL = exp(sum(NLL) / sum(tokens))
# Bits-per-byte: BPB = log2(PPL) / 4  (assumes 4 chars/token)

def compute_perplexity(model, tokenizer, text):
    input_ids = tokenizer(text, return_tensors="pt").input_ids
    with torch.no_grad():
        outputs = model(input_ids, labels=input_ids)
        nll = outputs.loss.item() * input_ids.shape[1]
        num_tokens = input_ids.shape[1]
    return np.exp(nll / num_tokens)
\end{lstlisting}

\textbf{Context Impact Protocol:}
\begin{lstlisting}[language=Python]
# For each document:
# 1. Split at mid-point (50% tokenization)
# 2. Context = first half, Continuation = second half
# 3. Cold-start: PPL(continuation | no_context)
# 4. Warm-start: PPL(continuation | context)
# 5. Reduction % = (PPL_cold - PPL_warm) / PPL_cold * 100

def evaluate_context_impact(model, tokenizer, document):
    tokens = tokenizer(document).input_ids
    mid_point = len(tokens) // 2
    
    context_tokens = tokens[:mid_point]
    continuation_tokens = tokens[mid_point:]
    
    # Cold-start: no preceding context
    ppl_cold = compute_perplexity(
        model, tokenizer, 
        tokenizer.decode(continuation_tokens)
    )
    
    # Warm-start: with preceding context
    ppl_warm = compute_nll_conditional(
        model, tokenizer,
        tokenizer.decode(context_tokens),
        tokenizer.decode(continuation_tokens)
    )
    
    reduction = (ppl_cold - ppl_warm) / ppl_cold
    return ppl_cold, ppl_warm, reduction
\end{lstlisting}

\textbf{Statistical Analysis:}
\begin{itemize}
\item Bootstrap 95\% confidence intervals (1000 iterations)
\item Paired t-tests (without vs with context)
\item Cohen's d effect sizes
\item Distribution analysis (skewness, kurtosis, percentiles)
\end{itemize}

\subsection{Metrics}

\textbf{Perplexity Metrics:}
\begin{itemize}
\item \textbf{Overall Perplexity:} $\text{PPL} = \exp\left(\frac{\sum \text{NLL}}{\sum \text{tokens}}\right)$
\item \textbf{Average Loss:} $\frac{\sum \text{NLL}}{\sum \text{tokens}}$
\item \textbf{Bits per Byte:} $\frac{\log_2(\text{PPL})}{4}$
\item \textbf{Sequence Statistics:} Mean, median, std, percentiles (5th, 25th, 75th, 95th)
\end{itemize}

\textbf{Distribution Analysis:}
\begin{itemize}
\item \textbf{Central Tendency:} Mean, median, 95\% CI
\item \textbf{Dispersion:} Standard deviation, variance, IQR, range
\item \textbf{Shape:} Skewness, kurtosis
\item \textbf{Extremes:} Min, max values
\end{itemize}

\textbf{Context Impact Metrics:}
\begin{itemize}
\item \textbf{Without Context:} Mean PPL, 95\% CI, median, std
\item \textbf{With Context:} Mean PPL, 95\% CI, median, std
\item \textbf{Reduction:} Mean, 95\% CI, percent, median
\item \textbf{Significance:} t-statistic, p-value, effect size (Cohen's d)
\end{itemize}

\subsection{Analysis}

\subsubsection{Key Findings}

\paragraph{Strengths of AWQ:}
\begin{itemize}
    \item \textbf{Commonsense Reasoning:} AWQ achieves the best or competitive performance across all commonsense benchmarks:
    \begin{itemize}
        \item Hellaswag: 0.694 (acc\_norm), close to FP16 baseline (0.72)
        \item WinoGrande: 0.766, outperforming GPTQ (0.752) and NF4 (0.68)
        \item PIQA: 0.848 (acc\_norm), highest among all quantized models
        \item ARC-Easy: 0.784, strongest performance
        \item ARC-Challenge: 0.530 (acc\_norm), competitive with GPTQ (0.510)
    \end{itemize}
    \item \textbf{Binary Classification:} BoolQ accuracy of 0.846 significantly exceeds all other models, suggesting AWQ preserves decision boundaries well for simpler classification tasks.
\end{itemize}

\paragraph{Weaknesses of AWQ:}
\begin{itemize}
    \item \textbf{Reading Comprehension:} Substantial degradation on SQuADv2:
    \begin{itemize}
        \item Overall Exact Match: 0.255 vs GPTQ's 0.53 (52\% relative drop)
        \item Cannot identify unanswerable questions (NoAns F1 = 0.00)
        \item Suggests issues with abstention and nuanced answer extraction
    \end{itemize}
    \item \textbf{Complex Reasoning:} Severe failure on DROP:
    \begin{itemize}
        \item Exact Match: 0.00 (unable to produce any exact answers)
        \item F1: 0.027 (96\% drop from baseline)
        \item Indicates AWQ struggles with multi-hop reasoning and numerical operations
    \end{itemize}
\end{itemize}
